Messages Hidden in Matrices

Nested Lists, Matrix Transformations & Phrase Decoding

Project context

Explore a phrase-encoding system built around binary character matrices. The implementation validates matrix structure, combines and transforms rows, and supports rotation, reflection and phrase reconstruction.

Implementation highlights

  • Validate rectangular matrices and transform binary values.
  • Rotate and reflect nested-list representations.
  • Compose helper functions into a phrase-decoding workflow.

def rotate_matrix(matrix: list[list[Element]]) -> list[list[Element]]:
    if not is_valid_matrix(matrix):
        return matrix

    # handle empty matrix
    if len(matrix) == 0:
        return matrix

    # create new matrix for 90 degree clockwise rotation
    rows = len(matrix)
    cols = len(matrix[0])
    new_matrix = []

    # fill new matrix by picking elements in reverse row order for each column
    for c in range(cols):
        new_row = []
        for r in range(rows - 1, -1, -1):
            new_row.append(matrix[r][c])
        new_matrix.append(new_row)

    return new_matrix

def flip_matrix(
    matrix: list[list[Element]], horizontal: bool
) -> list[list[Element]]:
    if not is_valid_matrix(matrix):
        return matrix

    # create new matrix for flipping
    new_matrix = []

    if horizontal:
        # horizontal flip means flip top to bottom (reverse row order)
        for i in range(len(matrix) - 1, -1, -1):
            new_row = []
            for j in range(len(matrix[i])):
                new_row.append(matrix[i][j])
            new_matrix.append(new_row)
    else:
        # vertical flip means flip left to right (reverse column order)
        for row in matrix:
            new_row = []
            for j in range(len(row) - 1, -1, -1):
                new_row.append(row[j])
            new_matrix.append(new_row)

    return new_matrix
